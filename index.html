fbSecureClient.setInsecure(); // Overrides heavy SSL computations to protect memory heap String url = String(firebase_url) + "live.json"; http.begin(fbSecureClient, url); http.addHeader("Content-Type", "application/json"); http.PUT(body); http.end(); } } void checkFirebaseControls() { if (WiFi.status() == WL_CONNECTED && (millis() - lastFirebaseControlCheck > 1000)) { lastFirebaseControlCheck = millis(); HTTPClient http; StaticJsonDocument<128> doc; deserializeJson(doc, payload); int hitachiCmd = doc["cmd_hitachi"] | 0; int kaveriCmd = doc["cmd_kaveri"] | 0; if (hitachiCmd == 1) { Serial.println("✓ Firebase Web Trigger: Executing Hitachi Pump Relay Pulse!"); digitalWrite(RELAY_HITACHI_PIN, HIGH); delay(1000); // 1 second hardware pulse time digitalWrite(RELAY_HITACHI_PIN, LOW); http.PUT("{\"cmd_hitachi\":0,\"cmd_kaveri\":0}"); // Reset trigger switches back to 0 } else if (kaveriCmd == 1) { Serial.println("✓ Firebase Web Trigger: Executing Kaveri Pump Relay Pulse!"); digitalWrite(RELAY_KAVERI_PIN, HIGH); delay(1000); // 1 second hardware pulse time digitalWrite(RELAY_KAVERI_PIN, LOW); http.PUT("{\"cmd_hitachi\":0,\"cmd_kaveri\":0}"); // Reset trigger switches back to 0 } } http.end(); } } // ========================================================================= // 4.
